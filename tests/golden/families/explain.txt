# explain.txt — golden-parity family for the EXPLAIN forms of the DDL, backup,
# and restore commands plus the seeded EXPLAIN INSERT/REPLACE/TTL forms (phase
# 2i, AI-2i-B; consolidated into the single golden tree in phase 2j, AI-2j-5).
#
# The original ddbsh never golden-tested these EXPLAIN forms. Per the command-
# coverage matrix (design doc 2026-10-21-P2i-01, AI-2i-A), `EXPLAIN <cmd>` is a
# distinct command from `<cmd>` (D-2i-3) and is covered separately. These are
# the EXPLAIN forms whose serialized request is deterministic and backend-free
# (pure request serialization — no DescribeTable / DescribeTTL probe is needed
# to plan them), so they produce a stable golden with no live resources and no
# normalization:
#
#   EXPLAIN CREATE TABLE  (on-demand default, provisioned, with a GSI)
#   EXPLAIN DROP TABLE
#   EXPLAIN ALTER TABLE   (billing mode on-demand / provisioned, PITR on/off)
#   EXPLAIN BACKUP TABLE ... CALLED
#   EXPLAIN DROP BACKUP
#   EXPLAIN DESCRIBE BACKUP
#   EXPLAIN RESTORE TABLE ... FROM BACKUP
#   EXPLAIN RESTORE TABLE ... FROM ... TO PITR
#
# The expected outputs were RECORDED FROM THE ORIGINAL ddbsh (the production
# `ddbsh-old` oracle binary, driven through the golden harness, D-2i-2) and are
# reproduced byte-for-byte by the reimplementation. EXPLAIN output is
# deterministic (no sort, no -s).
#
# EXPLAIN INSERT / EXPLAIN REPLACE are NOT here: their planner resolves the
# table key schema via DescribeTable (same as EXPLAIN SELECT, arch §11), so they
# need a seeded table and live in a seeded family. EXPLAIN SELECT /
# EXPLAIN UPDATE / EXPLAIN DELETE / EXPLAIN transaction are already covered by
# the isel-/iut-/delete/itxt- families.
#
# ${RD} is the single golden results dir (tests/golden/results), set by
# tests/golden/CMakeLists.txt.

add_test(NAME iexp-create-table COMMAND ddbshtest -q -T ${RD}/iexp-create-table.exp -c "explain create table foo ( pk number, rk number ) primary key (pk hash, rk range);")
add_test(NAME iexp-create-table-provisioned COMMAND ddbshtest -q -T ${RD}/iexp-create-table-provisioned.exp -c "explain create table foo ( pk number ) primary key (pk hash) billing mode provisioned ( 5 rcu, 10 wcu );")
add_test(NAME iexp-create-table-gsi COMMAND ddbshtest -q -T ${RD}/iexp-create-table-gsi.exp -c "explain create table foo ( pk number, gsipk number ) primary key (pk hash) gsi (myidx on (gsipk hash) projecting all);")
add_test(NAME iexp-drop-table COMMAND ddbshtest -q -T ${RD}/iexp-drop-table.exp -c "explain drop table foo;")
add_test(NAME iexp-alter-billing-ondemand COMMAND ddbshtest -q -T ${RD}/iexp-alter-billing-ondemand.exp -c "explain alter table foo set billing mode on demand;")
add_test(NAME iexp-alter-billing-provisioned COMMAND ddbshtest -q -T ${RD}/iexp-alter-billing-provisioned.exp -c "explain alter table foo set billing mode provisioned ( 5 rcu, 10 wcu );")
add_test(NAME iexp-alter-pitr-enabled COMMAND ddbshtest -q -T ${RD}/iexp-alter-pitr-enabled.exp -c "explain alter table foo set pitr enabled;")
add_test(NAME iexp-alter-pitr-disabled COMMAND ddbshtest -q -T ${RD}/iexp-alter-pitr-disabled.exp -c "explain alter table foo set pitr disabled;")
add_test(NAME iexp-backup COMMAND ddbshtest -q -T ${RD}/iexp-backup.exp -c "explain backup table foo called \"mybackup\";")
add_test(NAME iexp-drop-backup COMMAND ddbshtest -q -T ${RD}/iexp-drop-backup.exp -c "explain drop backup \"arn:aws:dynamodb:us-east-1:632195519165:table/foo/backup/01234567890123-abcdef00\";")
add_test(NAME iexp-describe-backup COMMAND ddbshtest -q -T ${RD}/iexp-describe-backup.exp -c "explain describe backup \"arn:aws:dynamodb:us-east-1:632195519165:table/foo/backup/01234567890123-abcdef00\";")
add_test(NAME iexp-restore-backup COMMAND ddbshtest -q -T ${RD}/iexp-restore-backup.exp -c "explain restore table bar from backup \"arn:aws:dynamodb:us-east-1:632195519165:table/foo/backup/01234567890123-abcdef00\";")
add_test(NAME iexp-restore-pitr COMMAND ddbshtest -q -T ${RD}/iexp-restore-pitr.exp -c "explain restore table bar from foo to pitr \"2024-01-01T00:00:00Z\";")

# --- seeded EXPLAIN forms (INSERT / REPLACE / ALTER ... SET TTL) -------------
# Unlike the pure-serialization EXPLAIN DDL forms above, these EXPLAIN forms
# probe the live table before serializing:
#   * INSERT / REPLACE resolve the table's key schema via DescribeTable to build
#     the PutItem (INSERT adds the attribute_not_exists(#pk) condition; arch §7,
#     §8) -- the same reason EXPLAIN SELECT is not backend-free (arch §11).
#   * ALTER ... SET TTL runs a DescribeTTL probe (arch §7); EXPLAIN prints only
#     the resulting UpdateTimeToLive serialization.
# So these tests are SEEDED: each -c creates a real table (CREATE waits for
# ACTIVE), runs the EXPLAIN form(s), then drops it (self-contained lifecycle).
#
# WARNING: creates and drops REAL DynamoDB tables (ddbsh_iexp_insert,
# ddbsh_ttl) on each run (us-east-1). Self-cleaning.

add_test(NAME iexp-insert-replace COMMAND ddbshtest -q -T ${RD}/iexp-insert-replace.exp -c "create table ddbsh_iexp_insert ( pk number ) primary key (pk hash); explain insert into ddbsh_iexp_insert (pk, a) values (1, 2); explain replace into ddbsh_iexp_insert (pk, a) values (1, 2); drop table ddbsh_iexp_insert;")

add_test(NAME iexp-alter-ttl COMMAND ddbshtest -q -T ${RD}/iexp-alter-ttl.exp -c "create table ddbsh_ttl ( pk number ) primary key (pk hash); explain alter table ddbsh_ttl set ttl (expiry); drop table ddbsh_ttl;")
