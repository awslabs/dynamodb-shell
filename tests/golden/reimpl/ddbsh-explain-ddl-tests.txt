# ddbsh-explain-ddl-tests.txt — golden-parity family for the deterministic
# EXPLAIN forms of the DDL, backup, and restore commands (phase 2i, AI-2i-B).
#
# The original ddbsh never golden-tested these EXPLAIN forms. Per the command-
# coverage matrix (design doc 2026-10-21-P2i-01, AI-2i-A), `EXPLAIN <cmd>` is a
# distinct command from `<cmd>` (D-2i-3) and is covered separately. These are
# the EXPLAIN forms whose serialized request is deterministic and backend-free
# (pure request serialization — no DescribeTable / DescribeTTL probe is needed
# to plan them), so they produce a stable golden with no live resources and no
# normalization:
#
#   EXPLAIN CREATE TABLE  (on-demand default, provisioned, with a GSI)
#   EXPLAIN DROP TABLE
#   EXPLAIN ALTER TABLE   (billing mode on-demand / provisioned, PITR on/off)
#   EXPLAIN BACKUP TABLE ... CALLED
#   EXPLAIN DROP BACKUP
#   EXPLAIN DESCRIBE BACKUP
#   EXPLAIN RESTORE TABLE ... FROM BACKUP
#   EXPLAIN RESTORE TABLE ... FROM ... TO PITR
#
# The expected outputs were RECORDED FROM THE ORIGINAL ddbsh (the production
# `ddbsh-old` oracle binary, driven through the golden harness, D-2i-2) and are
# reproduced byte-for-byte by the reimplementation. EXPLAIN output is
# deterministic (no sort, no -s).
#
# EXPLAIN INSERT / EXPLAIN REPLACE are NOT here: their planner resolves the
# table key schema via DescribeTable (same as EXPLAIN SELECT, arch §11), so they
# need a seeded table and live in a seeded family. EXPLAIN SELECT /
# EXPLAIN UPDATE / EXPLAIN DELETE / EXPLAIN transaction are already covered by
# the isel-/iut-/delete/itxt- families.
#
# ${RD_REIMPL} is the reimplementation-owned golden results dir
# (tests/golden/reimpl/results), set by tests/golden/CMakeLists.txt.

add_test(NAME iexp-create-table COMMAND ddbshtest -q -T ${RD_REIMPL}/iexp-create-table.exp -c "explain create table foo ( pk number, rk number ) primary key (pk hash, rk range);")
add_test(NAME iexp-create-table-provisioned COMMAND ddbshtest -q -T ${RD_REIMPL}/iexp-create-table-provisioned.exp -c "explain create table foo ( pk number ) primary key (pk hash) billing mode provisioned ( 5 rcu, 10 wcu );")
add_test(NAME iexp-create-table-gsi COMMAND ddbshtest -q -T ${RD_REIMPL}/iexp-create-table-gsi.exp -c "explain create table foo ( pk number, gsipk number ) primary key (pk hash) gsi (myidx on (gsipk hash) projecting all);")
add_test(NAME iexp-drop-table COMMAND ddbshtest -q -T ${RD_REIMPL}/iexp-drop-table.exp -c "explain drop table foo;")
add_test(NAME iexp-alter-billing-ondemand COMMAND ddbshtest -q -T ${RD_REIMPL}/iexp-alter-billing-ondemand.exp -c "explain alter table foo set billing mode on demand;")
add_test(NAME iexp-alter-billing-provisioned COMMAND ddbshtest -q -T ${RD_REIMPL}/iexp-alter-billing-provisioned.exp -c "explain alter table foo set billing mode provisioned ( 5 rcu, 10 wcu );")
add_test(NAME iexp-alter-pitr-enabled COMMAND ddbshtest -q -T ${RD_REIMPL}/iexp-alter-pitr-enabled.exp -c "explain alter table foo set pitr enabled;")
add_test(NAME iexp-alter-pitr-disabled COMMAND ddbshtest -q -T ${RD_REIMPL}/iexp-alter-pitr-disabled.exp -c "explain alter table foo set pitr disabled;")
add_test(NAME iexp-backup COMMAND ddbshtest -q -T ${RD_REIMPL}/iexp-backup.exp -c "explain backup table foo called \"mybackup\";")
add_test(NAME iexp-drop-backup COMMAND ddbshtest -q -T ${RD_REIMPL}/iexp-drop-backup.exp -c "explain drop backup \"arn:aws:dynamodb:us-east-1:632195519165:table/foo/backup/01234567890123-abcdef00\";")
